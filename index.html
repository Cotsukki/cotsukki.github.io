let topicsCatalog = []; let currentTopicData = null; let currentActiveTopicId = null; let currentList = []; let currentIndex = 0; let currentFilter = 'all'; let progress = {}; const elements = { header: document.getElementById('topic-header'), statCorrect: document.getElementById('stat-correct'), statMissed: document.getElementById('stat-missed'), statLeft: document.getElementById('stat-left'), grid: document.getElementById('question-grid'), prompt: document.getElementById('q-prompt'), choices: document.getElementById('q-choices'), note: document.getElementById('q-note'), btnPrev: document.getElementById('btn-prev'), btnNext: document.getElementById('btn-next'), btnClear: document.getElementById('btn-clear'), filters: document.querySelectorAll('.filter-btn') }; async function init() { loadProgress(); setupEventListeners(); try { // 1. Fetch the catalog of topics const response = await fetch('topics.json'); topicsCatalog = await response.json(); // 2. Build the navigation bar renderTopicHeaders(); // 3. Load the first topic by default if the catalog isn't empty if (topicsCatalog.length > 0) { await loadTopic(topicsCatalog[0]); } } catch (error) { elements.prompt.textContent = "Error loading topics.json. Please ensure the file exists."; } } function setupEventListeners() { elements.btnPrev.addEventListener('click', () => navigate(-1)); elements.btnNext.addEventListener('click', () => navigate(1)); elements.btnClear.addEventListener('click', clearTopicProgress); elements.filters.forEach(btn => { btn.addEventListener('click', (e) => { elements.filters.forEach(b => b.classList.remove('active')); e.target.classList.add('active'); currentFilter = e.target.dataset.filter; applyFilter(); }); }); } function renderTopicHeaders() { elements.header.innerHTML = ''; topicsCatalog.forEach(topicMeta => { const btn = document.createElement('button'); btn.className = `topic-btn ${topicMeta.id === currentActiveTopicId ? 'active' : ''}`; btn.textContent = topicMeta.name; btn.addEventListener('click', () => loadTopic(topicMeta)); elements.header.appendChild(btn); }); } async function loadTopic(topicMeta) { // Don't re-fetch if we are already on this topic if (currentActiveTopicId === topicMeta.id) return; elements.prompt.textContent = `Loading ${topicMeta.name}...`; elements.choices.innerHTML = ''; elements.grid.innerHTML = ''; try { // Fetch only the specific subject file const response = await fetch(topicMeta.file); currentTopicData = await response.json(); currentActiveTopicId = topicMeta.id; // Ensure a progress tracking object exists for this specific topic if (!progress[currentTopicData.topic]) { progress[currentTopicData.topic] = {}; } // Update active state on the navigation bar renderTopicHeaders(); // Reset filters and display the new questions currentFilter = 'all'; elements.filters.forEach(b => b.classList.remove('active')); elements.filters[0].classList.add('active'); applyFilter(); } catch (error) { elements.prompt.textContent = `Error loading ${topicMeta.file}.`; } } function loadProgress() { const saved = localStorage.getItem('reviewAppProgress'); if (saved) { progress = JSON.parse(saved); } } function saveProgress() { localStorage.setItem('reviewAppProgress', JSON.stringify(progress)); updateStats(); renderGrid(); } function applyFilter() { if (!currentTopicData) return; const topicProgress = progress[currentTopicData.topic]; if (currentFilter === 'all') { currentList = [...currentTopicData.questions]; } else if (currentFilter === 'unanswered') { currentList = currentTopicData.questions.filter(q => !topicProgress[q.id]); } else if (currentFilter === 'missed') { currentList = currentTopicData.questions.filter(q => topicProgress[q.id] && !topicProgress[q.id].correct); } currentIndex = 0; updateStats(); renderGrid(); renderQuestion(); } function updateStats() { const topicProgress = progress[currentTopicData.topic]; let correct = 0, missed = 0, left = 0; currentTopicData.questions.forEach(q => { if (!topicProgress[q.id]) left++; else if (topicProgress[q.id].correct) correct++; else missed++; }); elements.statCorrect.textContent = `Correct: ${correct}`; elements.statMissed.textContent = `Missed: ${missed}`; elements.statLeft.textContent = `Left: ${left}`; } function renderGrid() { elements.grid.innerHTML = ''; const topicProgress = progress[currentTopicData.topic]; currentList.forEach((q, index) => { const btn = document.createElement('div'); btn.className = `grid-item ${index === currentIndex ? 'active' : ''}`; btn.textContent = index + 1; if (topicProgress[q.id]) { btn.classList.add(topicProgress[q.id].correct ? 'correct' : 'wrong'); } btn.addEventListener('click', () => { currentIndex = index; renderGrid(); renderQuestion(); }); elements.grid.appendChild(btn); }); } function renderQuestion() { if (currentList.length === 0) { elements.prompt.textContent = "No questions match the current filter."; elements.choices.innerHTML = ''; elements.note.style.display = 'none'; elements.btnPrev.disabled = true; elements.btnNext.disabled = true; return; } const q = currentList[currentIndex]; const topicProgress = progress[currentTopicData.topic]; const answeredData = topicProgress[q.id]; elements.prompt.textContent = `${currentIndex + 1}. ${q.prompt}`; elements.choices.innerHTML = ''; q.choices.forEach(choice => { const btn = document.createElement('button'); btn.className = 'choice-btn'; btn.textContent = `${choice.letter}. ${choice.text}`; if (answeredData) { btn.disabled = true; if (choice.letter === answeredData.selected) { btn.classList.add(answeredData.correct ? 'selected-correct' : 'selected-wrong'); } if (!answeredData.correct && choice.letter === q.answer) { btn.classList.add('actual-correct'); } } else { btn.addEventListener('click', () => handleAnswer(q.id, choice.letter, q.answer)); } elements.choices.appendChild(btn); }); if (answeredData && q.note) { elements.note.textContent = q.note; elements.note.style.display = 'block'; } else { elements.note.style.display = 'none'; } elements.btnPrev.disabled = currentIndex === 0; elements.btnNext.disabled = currentIndex === currentList.length - 1; } function handleAnswer(qId, selected, correctLetter) { const isCorrect = selected === correctLetter; progress[currentTopicData.topic][qId] = { selected: selected, correct: isCorrect }; saveProgress(); renderQuestion(); } function navigate(dir) { currentIndex += dir; renderGrid(); renderQuestion(); } function clearTopicProgress() { if (confirm(`Are you sure you want to clear your progress for ${currentTopicData.topic}?`)) { progress[currentTopicData.topic] = {}; saveProgress(); applyFilter(); } } // Start the app init();
